// ══ CONNECTION ENGINE — Google Apps Script Relay ════════════════════════════
// Uses a user-deployed Apps Script web app as a CORS-free data bridge.
// This works from file://, localhost, anywhere — permanently.

let GAS_URL = ""; // Set by user in CONFIG

function extractSheetId(raw) {
  const v = raw.trim();
  const m = v.match(/\/spreadsheets\/d\/([a-zA-Z0-9_-]{20,})/);
  if (m) return m[1];
  if (!v.includes('/') && !v.startsWith('http') && v.length > 20) return v;
  return null;
}

async function fetchTabGAS(tabName) {
  const url = GAS_URL + "?tab=" + encodeURIComponent(tabName) + "&t=" + Date.now();
  const r = await fetch(url);
  if (!r.ok) throw new Error("HTTP " + r.status + " for " + tabName);
  const data = await r.json();
  if (data.error) throw new Error(data.error + " (tab: " + tabName + ")");
  return data.rows || [];
}

async function loadSheets(gasUrl) {
  showLoading();
  GAS_URL = gasUrl;
  const tabs = ["RC_Rigs","DD_Rigs","Compressors","LightVehicles","WorkOrders","Spares","KPITargets","Technicians"];
  const errors = [];
  async function safe(tab) {
    try { return await fetchTabGAS(tab); }
    catch(e) { errors.push(e.message); return []; }
  }
  const [rc,dd,comp,lv,wo,sp,kp,te] = await Promise.all(tabs.map(t => safe(t)));
  D = {rc, dd, comp, lv, wo, spares:sp, kpis:kp, techs:te, costs:[]};
  localStorage.setItem("smartis-gas-url", gasUrl);
  if (errors.length > 0) {
    const loaded = [rc,dd,comp,lv,wo,sp,kp,te].filter(x=>x.length>0).length;
    if (loaded === 0) {
      showError(
        "<strong>Apps Script returned errors for all tabs.</strong><br><br>" +
        "Errors: " + errors.slice(0,3).join(" | ") +
        "<br><br>Make sure you:<br>" +
        "1. Deployed the Apps Script as a web app<br>" +
        "2. Set <strong>Execute as: Me</strong><br>" +
        "3. Set <strong>Who has access: Anyone</strong><br>" +
        "4. Pasted the correct deployment URL (ends with /exec)<br><br>" +
        "<button class='btn-p' onclick='showSetup()'>⚙ RECONFIGURE</button>"
      );
      return;
    }
    const lr = $("last-refresh");
    if (lr) lr.textContent = "⚠ " + loaded + "/8 tabs loaded — " + new Date().toLocaleTimeString();
  } else {
    const lr = $("last-refresh");
    if (lr) lr.textContent = "✓ All 8 tabs loaded — " + new Date().toLocaleTimeString();
  }
  const dl=$("data-label"), ld=$("live-dot");
  if(dl){dl.textContent="LIVE DATA"; dl.style.color="var(--green)";}
  if(ld) ld.style.background="var(--green)";
  renderAll();
}

function testConnection() {
  const raw = $("sheet-id-input").value.trim();
  const res = $("test-result");
  if (!raw) { res.innerHTML='<span style="color:#f5a800;">⚠ Paste your Apps Script URL first.</span>'; return; }
  if (!raw.includes("script.google.com")) {
    res.innerHTML='<span style="color:#ff3355;">✕ This should be your Apps Script deployment URL — it must contain <strong>script.google.com</strong></span>';
    return;
  }
  res.innerHTML='<span style="color:#00c8f0;">🔄 Testing connection...</span>';
  fetch(raw + "?tab=RC_Rigs&t=" + Date.now())
    .then(r => r.json())
    .then(data => {
      if (data.error) {
        res.innerHTML='<span style="color:#ff3355;">✕ Script error: ' + data.error + '</span>';
      } else {
        const rows = data.rows || [];
        res.innerHTML='<span style="color:#00e087;">✓ Connected! RC_Rigs returned <strong>' + rows.length + ' row(s)</strong>.<br>Click <strong>CONNECT & LOAD</strong> to load all 8 tabs.</span>';
      }
    })
    .catch(e => {
      res.innerHTML='<span style="color:#ff3355;">✕ ' + e.message + '<br>Make sure you set <strong>Who has access: Anyone</strong> when deploying.</span>';
    });
}

// ══ COST TRACKER ════════════════════════════════════════════════════════════
function buildCostTracker(){
  const costs=D.costs||[];if(!$("cost-big"))return;if(!costs.length){const e=$("cost-big");if(e)e.innerHTML='<div style="grid-column:span 4;text-align:center;padding:20px;font-size:11px;color:var(--muted);">Add a Costs tab to your Google Sheet to enable cost tracking.</div>';return;}
  const totalBudget=costs.reduce((s,c)=>s+Number(c.Budget||0),0);
  const totalActual=costs.reduce((s,c)=>s+Number(c.Actual||0),0);
  const variance=totalActual-totalBudget;
  const overruns=costs.filter(c=>Number(c.Actual)>Number(c.Budget)).length;
  const cp=$("cost-period-badge");if(cp)cp.textContent="PERIOD: "+(costs[0]?.Period||"Current");

    {l:"Total Budget",v:"$"+totalBudget.toLocaleString(),c:"#00c8f0"},
